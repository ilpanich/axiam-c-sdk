cmake_minimum_required(VERSION 3.16)

# NOTE: CMake rejects pre-release semver suffixes, so the project version is the
# plain 1.0.0 triple. The alpha spelling ("1.0.0-alpha44") lives in vcpkg.json,
# conanfile.py, README and the AXIAM_VERSION macro.
project(axiam_c_sdk
    VERSION 1.0.0
    DESCRIPTION "AXIAM C SDK — REST client conforming to CONTRACT.md §1–§7, §9–§11, §13, §23 (incl. §6.1 mTLS)"
    HOMEPAGE_URL "https://github.com/ilpanich/axiam-c-sdk"
    LANGUAGES C)

# C11 is the DECLARED FLOOR: the standard the SDK promises to compile under, and
# the one every consumer's toolchain is guaranteed to accept. It is deliberately
# not the newest — raising it would exclude embedded and long-lived-distro
# toolchains for no gain the SDK actually needs.
#
# Overridable from the command line (`-DCMAKE_C_STANDARD=23`) rather than
# hardcoded, so CI can compile these exact sources under a newer standard as
# well. That is not a stylistic nicety: newer C standards REMOVE things and
# tighten others — K&R declarations are gone in C23, `bool`/`true`/`false`
# become keywords, and an implicit function declaration is an error rather than
# a warning. Code that is clean at C11 and broken at C23 compiles silently here
# and then fails in the build of the first consumer whose project sets a newer
# standard, which is entirely their prerogative and nothing to do with them.
#
# `if(NOT DEFINED ...)` rather than a CACHE entry so the default is unchanged
# for every existing consumer: pass nothing and this is still a C11 build.
if(NOT DEFINED CMAKE_C_STANDARD)
    set(CMAKE_C_STANDARD 11)
endif()
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
message(STATUS "AXIAM C SDK: compiling against C${CMAKE_C_STANDARD}")

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
endif()

option(AXIAM_BUILD_TESTS "Build the AXIAM C SDK test suite" OFF)
option(AXIAM_BUILD_EXAMPLES "Build the AXIAM C SDK example programs" ON)
option(AXIAM_ENABLE_COVERAGE "Instrument for gcov/lcov coverage" OFF)
option(AXIAM_BUILD_SHARED "Build the shared library variant" ON)
option(AXIAM_BUILD_STATIC "Build the static library variant" ON)

# Expose POSIX + GNU symbols (strdup, strncasecmp, pthread_barrier, usleep,
# strcasestr) while keeping the language standard at C11.
add_compile_definitions(_GNU_SOURCE)

include(GNUInstallDirs)

find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)

# ---- vendored cJSON (offline-friendly JSON) ----
add_library(axiam_cjson OBJECT third_party/cjson/cJSON.c)
target_include_directories(axiam_cjson PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party/cjson>)
set_target_properties(axiam_cjson PROPERTIES POSITION_INDEPENDENT_CODE ON)

set(AXIAM_SOURCES
    src/error.c
    src/util.c
    src/sensitive.c
    src/config.c
    src/json_build.c
    src/transport_curl.c
    src/client.c
    src/retry.c
    src/memo.c
    src/telemetry.c
    src/jwks.c
    src/guard.c
    src/webhook.c
    src/uma.c
    src/oidc.c
    src/oidc_validate.c
    src/oidc_refresh.c
    src/oidc_device.c
    src/oidc_exchange.c
    src/oidc_logout.c
    src/oidc_par.c
    src/reactor.c
    src/webauthn.c
    src/account.c
    src/opaque.c
    src/version.c
    # CONTRACT.md §27 management API. management.c is hand-written (transport, paging,
    # scope, the rule 7 classification); the other two are generated by
    # scripts/gen_management.py and their being committed is what keeps a consumer from
    # needing Python to build this library.
    src/management.c
    src/management_models.c
    src/management_ops.c)

# Compile the SDK sources once into an object library shared by both variants.
add_library(axiam_obj OBJECT ${AXIAM_SOURCES})
set_target_properties(axiam_obj PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(axiam_obj PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/cjson)
# ${CMAKE_DL_LIBS} is dlopen/dlsym: src/opaque.c resolves libaxiam_opaque_ffi at
# RUN time rather than link time (CONTRACT.md §23.1), so a consumer who never
# uses OPAQUE is not made to link the shared library — and axiam_opaque_available()
# can honestly answer 0. Empty on glibc >= 2.34, where libdl folded into libc.
target_link_libraries(axiam_obj PUBLIC CURL::libcurl OpenSSL::Crypto OpenSSL::SSL Threads::Threads ${CMAKE_DL_LIBS})

if(AXIAM_ENABLE_COVERAGE)
    target_compile_options(axiam_obj PRIVATE --coverage -O0 -g)
    target_link_options(axiam_obj PUBLIC --coverage)
endif()

set(AXIAM_INSTALL_TARGETS "")

# Shared library — the canonical `axiam` target.
if(AXIAM_BUILD_SHARED)
    add_library(axiam SHARED
        $<TARGET_OBJECTS:axiam_obj> $<TARGET_OBJECTS:axiam_cjson>)
    set_target_properties(axiam PROPERTIES
        VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR}
        OUTPUT_NAME axiam)
    target_include_directories(axiam PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    target_link_libraries(axiam PUBLIC CURL::libcurl OpenSSL::Crypto OpenSSL::SSL Threads::Threads ${CMAKE_DL_LIBS})
    if(AXIAM_ENABLE_COVERAGE)
        target_link_options(axiam PUBLIC --coverage)
    endif()
    add_library(axiam::axiam ALIAS axiam)
    list(APPEND AXIAM_INSTALL_TARGETS axiam)
endif()

# Static library variant (OUTPUT_NAME axiam so it produces libaxiam.a).
if(AXIAM_BUILD_STATIC)
    add_library(axiam_static STATIC
        $<TARGET_OBJECTS:axiam_obj> $<TARGET_OBJECTS:axiam_cjson>)
    set_target_properties(axiam_static PROPERTIES OUTPUT_NAME axiam)
    target_include_directories(axiam_static PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    target_link_libraries(axiam_static PUBLIC CURL::libcurl OpenSSL::Crypto OpenSSL::SSL Threads::Threads ${CMAKE_DL_LIBS})
    if(NOT AXIAM_BUILD_SHARED)
        add_library(axiam::axiam ALIAS axiam_static)
    endif()
    list(APPEND AXIAM_INSTALL_TARGETS axiam_static)
endif()

# ---- install rules + package config ----
install(TARGETS ${AXIAM_INSTALL_TARGETS}
    EXPORT axiam-c-sdkTargets
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/axiam DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

install(EXPORT axiam-c-sdkTargets
    FILE axiam-c-sdkTargets.cmake
    NAMESPACE axiam::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/axiam-c-sdk)

include(CMakePackageConfigHelpers)
configure_package_config_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/cmake/axiam-c-sdkConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/axiam-c-sdkConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/axiam-c-sdk)
write_basic_package_version_file(
    ${CMAKE_CURRENT_BINARY_DIR}/axiam-c-sdkConfigVersion.cmake
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion)
install(FILES
    ${CMAKE_CURRENT_BINARY_DIR}/axiam-c-sdkConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/axiam-c-sdkConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/axiam-c-sdk)

# ---- tests ----
if(AXIAM_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

# ---- examples ----
if(AXIAM_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()

# ---- CPack (.tar.gz) ----
set(CPACK_PACKAGE_NAME "axiam-c-sdk")
set(CPACK_PACKAGE_VENDOR "AXIAM")
set(CPACK_PACKAGE_VERSION "1.0.0-alpha44")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "${PROJECT_DESCRIPTION}")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_GENERATOR "TGZ")
set(CPACK_SOURCE_GENERATOR "TGZ")
include(CPack)
